## 🔢 **ОПЕРАТОРЫ**

```c
// Арифметика
a = b + c;    // Сложение
a = b - c;    // Вычитание
a = b * c;    // Умножение
a = b / c;    // Деление
a = b % c;    // Остаток от деления

// Присваивание
a += b;       // a = a + b
a -= b;       // a = a - b
a *= b;       // a = a * b
a /= b;       // a = a / b
a %= b;       // a = a % b
a++;          // a = a + 1
a--;          // a = a - 1

// Сравнение
a == b;       // Равно
a != b;       // Не равно
a > b;        // Больше
a < b;        // Меньше
a >= b;       // Больше или равно
a <= b;       // Меньше или равно

// Логика
a && b;       // И (оба истинны)
a || b;       // ИЛИ (хотя бы один истинен)
!a;           // НЕ (инверсия)

// Битовые (embedded-классика!)
a & b;        // Побитовое И
a | b;        // Побитовое ИЛИ
a ^ b;        // Побитовое XOR (исключающее ИЛИ)
~a;           // Побитовое НЕ (инверсия)
a << n;       // Сдвиг влево на n бит (умножение на 2^n)
a >> n;       // Сдвиг вправо на n бит (деление на 2^n)
```

    // выполнится 10 раз: i = 0,1,2...9

    // делаем, пока условие истинно

    // выполнится хотя бы 1 раз

for (int i = 0; i < 100; i++) {

## ⚡ **BIT MANIPULATION** (embedded-must-have!)

```c
// Установка бита (set bit)
REG |= (1 << n);          // установить бит n в 1

// Сброс бита (clear bit)
REG &= ~(1 << n);         // установить бит n в 0

// Переключение бита (toggle bit)
REG ^= (1 << n);          // инвертировать бит n

// Проверка бита (check bit)
if (REG & (1 << n)) {     // если бит n = 1
    // ...
}

// Маска битов
REG &= 0b11110000;        // оставить только старшие 4 бита
REG |= 0b00001111;        // установить младшие 4 бита в 1

// Сдвиг для записи
REG = (value << 4) & 0xF0;  // сдвинуть value на 4 бита влево и замаскировать
```

## 🎯 **EMBEDDED-СПЕЦИФИКА**

```c
// Работа с регистрами (пример для GPIO)
#define GPIO_PORT   (*(volatile uint32_t*)0x40021000)
#define GPIO_PIN_5  (1 << 5)

// Включить пин 5
GPIO_PORT |= GPIO_PIN_5;

// Выключить пин 5
GPIO_PORT &= ~GPIO_PIN_5;

// Переключить пин 5
GPIO_PORT ^= GPIO_PIN_5;

// Проверить состояние
if (GPIO_PORT & GPIO_PIN_5) {
    // пин в высоком уровне
}

// Макросы
#define SET_BIT(reg, bit)     ((reg) |= (1 << (bit)))
#define CLEAR_BIT(reg, bit)   ((reg) &= ~(1 << (bit)))
#define TOGGLE_BIT(reg, bit)  ((reg) ^= (1 << (bit)))
#define CHECK_BIT(reg, bit)   ((reg) & (1 << (bit)))
```
